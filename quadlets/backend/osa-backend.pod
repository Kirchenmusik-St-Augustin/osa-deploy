[Unit]
Wants=osa-backend.service osa-backend-pg.service osa-backend-valkey.service osa-backend-worker.service
Before=osa-backend.service osa-backend-pg.service osa-backend-valkey.service osa-backend-worker.service

[Pod]
PodName=osa-backend-pod
PublishPort=127.0.0.1:21000:8000
# PostgreSQL is deliberately not published on the host: every local account
# could reach the port. Diagnostics run inside the container:
#   podman exec -it osa-backend-pg psql -U <role> -d osa
# IPv4-only: pasta's default dual-stack IPv6 passthrough is fragile in this
# host's network setup and can silently black-hole outbound IPv6
# connections (e.g. to Google's cert endpoint during Google Login) without
# ever erroring or timing out on its own. Nothing in this stack needs
# outbound IPv6.
Network=pasta:-4

[Install]
WantedBy=default.target
