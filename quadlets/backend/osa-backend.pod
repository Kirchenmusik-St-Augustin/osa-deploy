[Unit]

Before=osa-backend.service osa-backend-pg.service

[Pod]
PodName=osa-backend-pod
PublishPort=127.0.0.1:21000:8000
# Loopback-only -- lets an admin run psql directly from the host for
# diagnostics, never reachable from outside.
PublishPort=127.0.0.1:5432:5432
# Maps the container's "app" user (uid=1000/gid=1000, see Dockerfile) onto
# this host's calling user -- without it, rootless Podman's default
# namespace mapping puts uid 1000 on an arbitrary subuid instead of onto
# the owner of the mounted data directories (e.g. the osa-backend-pg
# Postgres volume, mode 0700), so access would fail with a permission
# error.
UserNS=keep-id:uid=1000,gid=1000
# IPv4-only: pasta's default dual-stack IPv6 passthrough is fragile in this
# host's network setup and can silently black-hole outbound IPv6
# connections (e.g. to Google's cert endpoint during Google Login) without
# ever erroring or timing out on its own. Nothing in this stack needs
# outbound IPv6.
Network=pasta:-4

[Install]
WantedBy=default.target
